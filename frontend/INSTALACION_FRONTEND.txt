================================================================================
PASO 2: INSTALAR DEPENDENCIAS
================================================================================

cd /home/aplicacion/projects/adminapps/frontend

# Instalar Node.js dependencies
npm install

================================================================================
PASO 3: CREAR DIRECTORIOS DE LOGS
================================================================================

mkdir -p /home/aplicacion/projects/adminapps/frontend/logs

================================================================================
PASO 4: INICIAR CON PM2
================================================================================

# Iniciar en modo desarrollo
pm2 start ecosystem.config.js

# Guardar configuración de PM2
pm2 save

# Ver estado
pm2 list

# Ver logs
pm2 logs adminapps-frontend

================================================================================
PASO 5: VERIFICAR FUNCIONAMIENTO
================================================================================

# Verificar puerto
ss -tlnp | grep 3000

# Probar en navegador
# http://192.168.100.100:3000 (directo)
# http://adminapps.isosmart.local (vía Nginx - requiere /etc/hosts)

================================================================================
PASO 6: CONFIGURAR /etc/hosts (OPCIONAL - DESARROLLO LOCAL)
================================================================================

# En tu máquina local (no el servidor), agregar a /etc/hosts:
192.168.100.100 adminapps.isosmart.local
192.168.100.100 isosmart.local
192.168.100.100 landing.isosmart.local

================================================================================
COMANDOS ÚTILES PM2
================================================================================

# Ver estado
pm2 status

# Reiniciar
pm2 restart adminapps-frontend

# Ver logs en tiempo real
pm2 logs adminapps-frontend --lines 100

# Detener
pm2 stop adminapps-frontend

# Eliminar del PM2
pm2 delete adminapps-frontend

# Guardar configuración actual
pm2 save
